\newcommand*{\rom}[1]{\uppercase\expandafter{\romannumeral #1\relax}}

\newcommand{\tablespacing}{
    \setlength{\tabcolsep}{12pt}
    \renewcommand{\arraystretch}{1.5}
}

\newenvironment{ctable}[1][ht]{
    \begingroup
    \tablespacing

    % Suppress underfull/overfull box warnings locally inside tables \hbadness controls reporting of underfull hbox
    % badness; setting it % to 10000 suppresses those warnings. \vbadness does the same for vboxes.
    \hbadness=10000
    \vbadness=10000

    \begin{table}[#1]
    \centering
}{
    \end{table}
    \endgroup
}

\mdfdefinestyle{dpstyle}{
    linecolor=paletteblue,
    linewidth=1pt,
    roundcorner=6pt,
    innertopmargin=8pt,
    innerbottommargin=8pt,
    innerleftmargin=8pt,
    innerrightmargin=8pt,
    splittopskip=10pt,
}
\newmdenv[style=dpstyle]{dpbox}

% Parameter order:
% 1. Design Principle ID: DP1, DP2, ...
% 2. Design Principle Title
% 3. Implementer, Aim, User
% 4. Mechanism
% 5. Rationale
\newcommand{\designprinciple}[5]{%
    \begin{dpbox}
        \textbf{\textcolor{paletteblue}{#1: #2}}\\
        \setlength{\tabcolsep}{8pt}
        \renewcommand{\arraystretch}{1.2}
        \setlength{\LTpre}{0pt}
        \setlength{\LTpost}{0pt}
        \begin{xltabular}{\linewidth}{@{}l|X@{}}
            \textbf{Implementer, Aim, User} & #3 \\[4pt]
            \textbf{Mechanism} & #4 \\[4pt]
            \textbf{Rationale} & #5 \\
        \end{xltabular}
    \end{dpbox}
}
